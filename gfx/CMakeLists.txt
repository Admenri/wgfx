cmake_minimum_required(VERSION 3.30)

find_package(Python3 COMPONENTS Interpreter)

# Generate the WebGPU C-API -> gfx:: C++ bridge from the webgpu.yml IDL.
# The generated .cc implements each wgpu<Object><Method> C function by casting
# the WGPU handle to the matching gfx:: object and forwarding to its C++ method.
set(WGFX_WEBGPU_YML "${PROJECT_SOURCE_DIR}/third_party/webgpu-headers/webgpu.yml")
set(WGFX_BRIDGE_DIR "${CMAKE_CURRENT_BINARY_DIR}/gen")
set(WGFX_BRIDGE_SOURCE "${WGFX_BRIDGE_DIR}/webgpu_gfx_bridge.cc")
file(MAKE_DIRECTORY ${WGFX_BRIDGE_DIR})

message(STATUS "Generating webgpu gfx bridge from ${WGFX_WEBGPU_YML}")
message(STATUS "Output directory: ${WGFX_BRIDGE_DIR}")

execute_process(
  COMMAND ${Python3_EXECUTABLE} -B
  "${PROJECT_SOURCE_DIR}/tools/shell_generate.py"
  --yml "${WGFX_WEBGPU_YML}"
  --output "${WGFX_BRIDGE_DIR}"
  RESULT_VARIABLE WGFX_BRIDGE_RESULT
)

if (NOT WGFX_BRIDGE_RESULT EQUAL 0)
  message(FATAL_ERROR "Failed to generate webgpu gfx bridge: ${WGFX_BRIDGE_RESULT}")
endif ()

add_library(wgfx SHARED)

# gfx:: class sources (manually maintained). Only the C-API bridge below is
# auto-generated by tools/shell_generate.py.
file(GLOB WGFX_GFX_SOURCES CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/*.cc"
  "${CMAKE_CURRENT_SOURCE_DIR}/common/*.cc"
)
file(GLOB WGFX_GFX_HEADERS CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/*.h"
)

target_sources(wgfx PRIVATE
  ${WGFX_GFX_SOURCES}
  ${WGFX_GFX_HEADERS}
  ${WGFX_BRIDGE_SOURCE}
)

# Shared-library build: WGPU_EXPORT becomes dllexport when building the DLL
# (WGPU_IMPLEMENTATION) and dllimport for consumers (WGPU_SHARED_LIBRARY).
target_compile_definitions(wgfx
  PUBLIC WGPU_SHARED_LIBRARY
  PRIVATE WGPU_IMPLEMENTATION
)

# The generated bridge includes "webgpu.h" (webgpu-headers) and "gfx/gfx_main.h"
# (resolved via the project-wide include of ${PROJECT_SOURCE_DIR}).
target_include_directories(wgfx PUBLIC
  ${PROJECT_SOURCE_DIR}/third_party/webgpu-headers
)

target_link_libraries(wgfx PUBLIC
  volk::volk
  Vulkan::Headers
  VulkanMemoryAllocator
  webgpu-cpp-header
)

# volk must expose the platform WSI entry points the surface implementation
# loads through it (vkCreateWin32SurfaceKHR on Windows, XCB/Wayland on Linux).
if (WIN32)
  target_compile_definitions(volk PRIVATE VK_USE_PLATFORM_WIN32_KHR)
elseif (UNIX)
  target_compile_definitions(volk PRIVATE
    VK_USE_PLATFORM_XCB_KHR
    VK_USE_PLATFORM_XLIB_KHR
    VK_USE_PLATFORM_WAYLAND_KHR)
endif ()
